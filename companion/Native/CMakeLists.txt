cmake_minimum_required(VERSION 3.20)
project(warframe_native LANGUAGES C CXX)
add_library(warframe_native SHARED native.cpp memory.cpp vendor/lzma/LzmaDec.c)
target_compile_features(warframe_native PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(warframe_native PRIVATE /W4 /WX /O2 /EHsc)
  set_property(TARGET warframe_native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(warframe_native PRIVATE -Wall -Wextra -Werror -O3 -fvisibility=hidden)
endif()
include(CTest)
if(UNIX AND NOT APPLE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(OVERLAY REQUIRED IMPORTED_TARGET gtk+-3.0 webkit2gtk-4.1 gtk-layer-shell-0)
  add_library(warframe_overlay SHARED overlay_linux.cpp)
  target_compile_features(warframe_overlay PRIVATE cxx_std_20)
  target_link_libraries(warframe_overlay PRIVATE PkgConfig::OVERLAY)
  target_compile_options(warframe_overlay PRIVATE -Wall -Wextra -Werror -O2 -fvisibility=hidden)
endif()
if(BUILD_TESTING)
  add_executable(native_tests tests.cpp)
  target_compile_features(native_tests PRIVATE cxx_std_20)
  target_link_libraries(native_tests PRIVATE warframe_native)
  add_test(NAME native_regressions COMMAND native_tests)
endif()
